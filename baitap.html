Accordion đơn giản

Bấm vào tiêu đề để nội dung trượt xuống/lên — dùng classList.toggle() kết hợp max-height và transition trong CSS.

Accordion là thành phần giao diện gồm nhiều tiêu đề, mỗi tiêu đề che một đoạn nội dung. Khi bấm vào tiêu đề, nội dung tương ứng sẽ mở ra (trượt xuống) và các mục khác có thể tự đóng lại.
CSS đặt max-height: 0 và overflow: hidden cho phần thân. Khi thêm class .open, max-height tăng lên một giá trị đủ lớn; thuộc tính transition: max-height .45s tạo hiệu ứng trượt mượt mà.
classList.toggle('open') tự động thêm class nếu chưa có và gỡ bỏ nếu đã có — chỉ một dòng code là có thể đóng/mở mà không cần kiểm tra thủ công.
Xem mã nguồn JavaScript 
// TK5 — Accordion: classList.toggle + max-height/transition
document.querySelectorAll('.acc-header').forEach(btn => {
  btn.addEventListener('click', () => {
    btn.parentElement.classList.toggle('open');
  });
});

/* CSS */
.acc-body { max-height: 0; overflow: hidden;
           transition: max-height .45s ease; }
.acc-item.open .acc-body { max-height: 240px; }

Tạo chuỗi mật khẩu ngẫu nhiên

Hàm tạo chuỗi dài 8 ký tự, chỉ gồm chữ cái thường và chữ số. Bấm nút để sinh mới.

a1b2c3d4 8 ký tự · chữ thường + số
Xem mã nguồn JavaScript 
// TK7 — Tạo mật khẩu ngẫu nhiên 8 ký tự (chữ thường + số)
function taoMatKhau() {
  const kyTu = 'abcdefghijklmnopqrstuvwxyz0123456789';
  let kq = '';
  for (let i = 0; i < 8; i++) {
    const viTri = Math.floor(Math.random() * kyTu.length);
    kq += kyTu[viTri];
  }
  return kq;
}

Chuyển đổi màu Hex sang RGB

Nhập mã màu Hex (ví dụ #FF0000 hoặc #F00) — hàm trả về giá trị rgb(r, g, b) tương ứng.

rgb(255, 0, 0)
Xem mã nguồn JavaScript 
// TK8 — Hex sang RGB (hỗ trợ #RGB và #RRGGBB)
function hexSangRgb(hex) {
  hex = hex.replace('#', '');
  if (hex.length === 3)
    hex = hex.split('').map(c => c + c).join('');
  const r = parseInt(hex.substr(0, 2), 16);
  const g = parseInt(hex.substr(2, 2), 16);
  const b = parseInt(hex.substr(4, 2), 16);
  return `rgb(${r}, ${g}, ${b})`;
}

Phân trang danh sách

Danh sách 50 mục, hiển thị 10 mục mỗi trang. Bấm nút số để chuyển trang.

Đang hiển thị mục 1 – 10 / 50
Xem mã nguồn JavaScript 
// K4 — Phân trang: 50 mục, 10 mục/trang
const TONG = 50, MOI_TRANG = 10;
let trangHienTai = 1;

function hienThi(trang) {
  const bd = (trang - 1) * MOI_TRANG;
  list.innerHTML = '';
  for (let i = bd; i < bd + MOI_TRANG; i++) {
    const div = document.createElement('div');
    div.className = 'item';
    div.textContent = `Mục số ${i + 1}`;
    list.appendChild(div);
  }
}
// Số trang = Math.ceil(TONG / MOI_TRANG) = 5

// K6 — Lắng nghe keydown, dịch chuyển div theo ↑ / ↓
let y = 132; // vị trí ban đầu (px)
const BUOC = 20;

document.addEventListener('keydown', e => {
  if (e.key === 'ArrowUp')   y = Math.max(0, y - BUOC);
  if (e.key === 'ArrowDown') y = Math.min(264, y + BUOC);

  e.preventDefault(); // chặn cuộn trang
});

Tính chỉ số BMI và phân loại

Nhập cân nặng (kg) và chiều cao (m) — hàm tính BMI = cân nặng / chiều cao² và phân loại.

—
—
<18.5 Thiếu cân18.5–24.9 Bình thường25–29.9 Thừa cân≥30 Béo phì
Xem mã nguồn JavaScript 
// K8 — Tính BMI và phân loại
function tinhBMI(kg, m) {
  const bmi = kg / (m * m);
  let loai;
  if      (bmi < 18.5) loai = 'Thiếu cân';
  else if (bmi < 25)   loai = 'Bình thường';
  else                 loai = 'Thừa cân';
  return { bmi: bmi.toFixed(1), loai };
}
